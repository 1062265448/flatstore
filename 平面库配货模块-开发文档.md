## 1. 模块概览

### 功能定位

平面库配货模块管理仓库库存批次的全生命周期：从库存录入（手动/批量导入/AI 图像识别）→ 创建配货单 → 锁定库存 → 发货 → 完成发运。

### 核心实体

| 实体 | 说明 |
|------|------|
| **库存批次** (`InventoryStock`) | 每批镍材的唯一批次记录，包含品级、重量、片数等 |
| **配货单** (`DistributionOrder`) | 客户订单，关联多个库存批次明细 |
| **客户** (`Customer`) | 配货单的接收方 |
| **配货明细** (`DistributionOrderItem`) | 配货单与库存批次的多对多关联 |
| **AI 识别历史** (`AiRecognitionHistory`) | 记录每次 AI 图像识别的结果与状态 |

### 技术栈（后端）

- **框架**: NestJS 10
- **ORM**: Prisma 6.19.3（**锁定版本，Prisma 7 不兼容当前 MySQL 适配器**）
- **数据库**: MySQL 8.0
- **AI**: 阿里云千问 qwen-vl-plus (图像识别)
- **验证**: class-validator + class-transformer
- **文件上传**: Multer

```prisma
// 客户
model Customer {
  id        Int       @id @default(autoincrement())
  name      String    @db.VarChar(100)
  contact   String?   @db.VarChar(50)
  phone     String?   @db.VarChar(20)
  address   String?   @db.VarChar(200)
  remark    String?   @db.Text  // ⚠️ Schema和DTO必须同时定义
  deletedAt DateTime? @map("deleted_at")
  createdAt DateTime  @default(now()) @map("created_at")
  updatedAt DateTime  @updatedAt @map("updated_at")

  orders DistributionOrder[]

  @@map("distribution_customer")
  @@index([name])
}

// 库存批次
model InventoryStock {
  id              Int       @id @default(autoincrement())
  packageNo       String?   @map("package_no") @db.VarChar(50)  // ⚠️ 用VARCHAR不用INT
  batchNo         String    @map("batch_no") @db.VarChar(50)
  grade           String    @db.VarChar(20)
  productType     String?   @map("product_type") @db.VarChar(50)
  specification   String?   @db.VarChar(50)
  weight          Decimal   @default(0) @db.Decimal(10, 3)
  pieceCount      Int       @default(0) @map("piece_count")
  location        String?   @db.VarChar(100)
  nickelContent   Decimal?  @map("nickel_content") @db.Decimal(5, 2)
  impurityContent String?  @map("impurity_content") @db.VarChar(100)
  status          String    @default("available") @db.VarChar(20)
  inspectionDate  DateTime? @map("inspection_date") @db.Date
  certificateNo   String?   @map("certificate_no") @db.VarChar(50)
  remark          String?   @db.Text
  sourceType      String?   @map("source_type") @db.VarChar(20)
  sourceImage     String?   @map("source_image") @db.VarChar(200)
  createdAt       DateTime  @default(now()) @map("created_at")
  updatedAt       DateTime  @updatedAt @map("updated_at")

  orderItems DistributionOrderItem[]

  @@map("inventory_stock")
  @@index([batchNo])
  @@index([grade])
  @@index([status])
  @@index([status, grade])  // ⚠️ 高频组合查询
}

// 配货单
model DistributionOrder {
  id           Int       @id @default(autoincrement())
  orderNo      String    @unique @map("order_no") @db.VarChar(50)
  customerId   Int       @map("customer_id")
  customerName String?   @map("customer_name") @db.VarChar(100)
  productSpec  String?   @map("product_spec") @db.VarChar(100)
  targetGrade  String?   @map("target_grade") @db.VarChar(20)
  totalWeight  Decimal?  @map("total_weight") @db.Decimal(10, 3)
  totalPieces  Int?      @map("total_pieces")
  status       String    @default("draft") @db.VarChar(20)
  shippedAt    DateTime? @map("shipped_at")
  driverName   String?   @map("driver_name") @db.VarChar(50)
  vehicleNo    String?   @map("vehicle_no") @db.VarChar(20)
  remark       String?   @db.Text
  deletedAt    DateTime? @map("deleted_at")
  createdAt    DateTime  @default(now()) @map("created_at")
  updatedAt    DateTime  @updatedAt @map("updated_at")

  customer Customer                @relation(fields: [customerId], references: [id])
  items    DistributionOrderItem[]

  @@map("distribution_order")
  @@index([orderNo])
  @@index([status])
  @@index([customerId])
  @@index([customerId, deletedAt, status])  // ⚠️ 高频组合查询
}

// 配货单明细
model DistributionOrderItem {
  id         Int      @id @default(autoincrement())
  orderId    Int      @map("order_id")
  stockId    Int      @map("stock_id")
  weight     Decimal  @default(0) @db.Decimal(10, 3)
  pieceCount Int      @default(0) @map("piece_count")
  createdAt  DateTime @default(now()) @map("created_at")

  order DistributionOrder @relation(fields: [orderId], references: [id], onDelete: Cascade)
  stock InventoryStock    @relation(fields: [stockId], references: [id], onDelete: Cascade)

  @@map("distribution_order_item")
  @@index([orderId])
  @@index([stockId])
}

// AI 识别历史
model AiRecognitionHistory {
  id           Int       @id @default(autoincrement())
  imageUrl     String    @map("image_url") @db.VarChar(200)
  result       String?   @db.Text           // ⚠️ 失败时可为null
  itemCount    Int       @map("item_count")
  status       String    @default("success") @db.VarChar(20)
  errorMessage String?   @map("error_message") @db.VarChar(500)
  batchNo      String?   @map("batch_no") @db.VarChar(50)
  grade        String?   @db.VarChar(20)
  date         DateTime? @db.Date
  createdAt    DateTime  @default(now()) @map("created_at")

  @@map("ai_recognition_history")
  @@index([status])
  @@index([createdAt(sort: Desc)])
}
```

#### 3.3.1 库存管理

| 方法 | 逻辑 | ⚠️ 注意事项 |
|------|------|-----------|

#### 3.3.2 配货单管理（状态机 + 事务）

**状态流转**:

```
draft ──confirm──→ confirmed ──ship──→ shipping ──deliver──→ shipped
  │                                                                 │
  └──────────────────── cancel ─────────────────────────────────────┘
  └────────────── cancel ──────┘
```

| 操作 | 事务内容 | ⚠️ 注意事项 |
|------|----------|-----------|
| **创建配货单** | ① 校验库存可用性 ② 创建订单+明细 ③ 锁定库存（status→'reserved'） | 批量查询库存（避免N+1） |
| **确认** | 状态 `draft` → `confirmed` | 仅限 draft |
| **发货** | 状态 `confirmed` → `shipping`，记录司机/车牌 | 仅限 confirmed |
| **完成发运** | 状态 `shipping` → `shipped`，库存 `reserved` → `shipped` | 仅限 shipping |
| **取消** | 释放库存 `reserved` → `available` | 仅限 draft/confirmed/shipping |
| **删除** | 释放库存 → 软删除（`deletedAt = NOW`） | 仅限 shipped/cancelled |

### 3.5 QwenAIService

### 6.2 AI 识别成功的关键保障（7 层防线）

#### 防线 1: Prompt 工程（千问 VL 专用优化）

```
你是一个票据识别专家。请分析这张图片中的表格数据。

要求：
1. 识别图片左上角"品名"右侧的产品类型文字（如电解镍、电积镍等）
2. 提取表格中的数据行，排除合计行/汇总行
3. 所有行的 productType 应为同一个值
4. 无法识别的字段请设为 null

返回 JSON 数组，包含：
- packageNo: 包号（数字）
- pieceCount: 片数（数字）
- netWeight: 净重/重量（数字）
- grade: 品级（如 Ni9996, Ni9990 等）
- productType: 产品类型
- batchNo: 批号
- date: 日期（格式 YYYY-MM-DD）
```

- ✅ "只返回 JSON 数组" → 减少解析失败
- ✅ "排除合计行/汇总行" → 避免无效数据

### 8.10 交互规范

| 场景 | 交互 | ⚠️ 注意 |
|------|------|---------|
| 列表操作 | 右键菜单 / 行内悬浮按钮 | |
| 新建/编辑 | 右侧抽屉 (Drawer) | 不跳转页面 |
| 删除确认 | Modal + 二次确认 | 显示影响范围 |
| AI 上传 | 拖拽区域 → 实时进度 → 结果预览确认 → 一键入库 | ⚠️ 必须有加载提示 |
| 配货创建 | 多步向导 (Step Form) | 选客户 → 选库存 → 汇总确认 |
| 批量操作 | 顶部操作栏，显示已选数量 | |
| 空状态 | 插画 + 引导文案 + 快捷操作按钮 | |
| 加载状态 | 骨架屏 (Skeleton) | ⚠️ 非 Loading Spinner |
| 分页 | `@current-change` + `@size-change` | ⚠️ 推荐分拆写法（更灵活），`@pagination` 仍可用但建议迁移 |

### 8.12 统计卡片组件 (StatCard.vue)

> Dashboard 核心组件，用于展示关键业务指标。

```vue
<!-- components/business/StatCard.vue -->
<template>
  <div
    class="stat-card"
    :class="{ 'stat-card--clickable': clickable, 'stat-card--loading': loading }"
    @click="handleClick"
  >
    <div class="stat-card__icon" :style="{ background: iconBg }">
      <el-icon :size="24" :color="iconColor">
        <component :is="icon" />
      </el-icon>
    </div>

    <div class="stat-card__content">
      <div class="stat-card__label">{{ label }}</div>
      <div class="stat-card__value">
        <span v-if="!loading" class="value-number" :class="{ 'value-number--animated': animate }">
          {{ formattedValue }}
        </span>
        <el-skeleton v-else :rows="0" style="width: 80px; height: 32px" />
        <span v-if="unit && !loading" class="value-unit">{{ unit }}</span>
      </div>
      <div v-if="trend !== undefined && !loading" class="stat-card__trend" :class="trendClass">
        <el-icon size="14">
          <component :is="trend >= 0 ? 'Top' : 'Bottom'" />
        </el-icon>
        <span>{{ Math.abs(trend) }}%</span>
        <span class="trend-label">较上周</span>
      </div>
    </div>

    <div v-if="$slots.extra" class="stat-card__extra">
      <slot name="extra" />
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { Top, Bottom } from '@element-plus/icons-vue'

const props = withDefaults(
  defineProps<{
    icon: string
    label: string
    value: number | string
    unit?: string
    iconColor?: string
    trend?: number
    clickable?: boolean
    loading?: boolean
    animate?: boolean
  }>(),
  {
    iconColor: '#6366F1',
    clickable: false,
    loading: false,
    animate: true,
  }
)

const emit = defineEmits<{
  click: []
}>()

const formattedValue = computed(() => {
  if (typeof props.value === 'number') {
    if (props.value >= 10000) {
      return (props.value / 10000).toFixed(1) + '万'
    }
    return props.value.toLocaleString('zh-CN')
  }
  return props.value
})

const iconBg = computed(() => {
  const color = props.iconColor || '#6366F1'
  // 将十六进制颜色转换为 rgba
  const r = parseInt(color.slice(1, 3), 16)
  const g = parseInt(color.slice(3, 5), 16)
  const b = parseInt(color.slice(5, 7), 16)
  return `rgba(${r}, ${g}, ${b}, 0.1)`
})

const trendClass = computed(() => ({
  'is-up': props.trend !== undefined && props.trend > 0,
  'is-down': props.trend !== undefined && props.trend < 0,
}))

const handleClick = () => {
  if (props.clickable) {
    emit('click')
  }
}
</script>
```

### 9.1 API 响应格式规范

```typescript
// ✅ 统一响应格式
interface NestResponse<T> {
  success: boolean;    // true=成功，false=失败
  message?: string;    // 错误信息
  data: T;             // 数据（列表时为数组，不是 { items: [...] }）
  total?: number;      // 分页总数
  page?: number;
  pageSize?: number;
}

// ✅ Axios 拦截器处理
request.interceptors.response.use(response => {

// ✅ Store 中正确访问
const res = await getInventoryList(params)
```

```typescript
// ❌ 致命错误：手动设置 Content-Type
axios.post(url, formData, {

FileInterceptor('file', {  // ⚠️ 字段名是 'file'
  limits: { fileSize: 10 * 1024 * 1024 },

})
```

### 10.2 致命问题深度分析

#### 🔴 致命 #9: Store 数据绑定错误

**症状**: 所有列表（库存/订单/客户）永远显示为空，但后端数据正常。

**根因链路**:
```
后端返回: { data: [库存数组], total: 50, page: 1 }
        ↓
Axios 响应拦截器: return response.data  →  res = { data: [...], total: 50 }
        ↓
❌ 旧代码: inventoryList.value = res.data.items
   res.data 是数组 → res.data.items = undefined
        ↓
✅ 正确: inventoryList.value = res.data  （res.data 就是数组）
```

**为什么静默失败**:
- `res.data.items` 在 JS 中不报错（数组的 `.items` 属性 = `undefined`）
- 静默失败，用户看到空列表，误以为数据没保存

#### 🔴 致命 #27: axios multipart/form-data boundary

**症状**: 上传图片到后端，后端无法解析文件。

**根因**: 手动设置 `Content-Type: multipart/form-data` 会覆盖浏览器自动生成的 `boundary`。

```typescript
// frontend/src/types/api.ts

/**
 * ⚠️ NestJS 统一响应格式
 * 后端所有接口返回此格式：{ success: boolean, data: T, message?: string }
 */
export interface NestResponse<T> {
  success: boolean    // true=成功，false=失败
  data: T             // 数据
  message?: string    // 错误/提示信息
}

/**
 * ⚠️ 分页响应类型（列表专用）
 * data 直接是数组，不是 { items: [...] }
 * 注意：NestJS 返回 { data: [...], total: N, page: N, pageSize: N }
 * Axios 拦截器返回 res.data 后，res.data 就是 { data: [...], total: N, ... }
 */
export interface PaginatedResponse<T> {
  data: T[]         // ⚠️ 直接是数组，不是 res.data.items
  total: number     // 总记录数
  page: number      // 当前页
  pageSize: number  // 每页条数
}

/**
 * AI 识别响应
 */
export interface AiRecognizeResponse {
  success: boolean
  data: AiRecognizeItem[]
  message: string
}

/**
 * AI 识别单条数据
 */
export interface AiRecognizeItem {
  packageNo?: number   // AI 返回数字

  grade?: string
  productType?: string
  batchNo?: string
```

```typescript
// utils/helpers.ts

/**
 * 格式化重量
 */
export const formatWeight = (value: string | number, unit: 'kg' | 't' = 'kg'): string => {
  const num = typeof value === 'string' ? parseFloat(value) : value
  if (isNaN(num)) return '0'

  if (unit === 't') {
    return `${(num / 1000).toFixed(3)} t`
  }
  return `${num.toFixed(3)} kg`
}

/**
 * 格式化日期
 */
export const formatDate = (date: string | Date, format: 'date' | 'datetime' = 'date'): string => {
  const d = typeof date === 'string' ? new Date(date) : date
  if (isNaN(d.getTime())) return '-'

  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  const hours = String(d.getHours()).padStart(2, '0')
  const minutes = String(d.getMinutes()).padStart(2, '0')

  if (format === 'datetime') {
    return `${year}-${month}-${day} ${hours}:${minutes}`
  }
  return `${year}-${month}-${day}`
}

/**
 * 生成唯一 ID
 */
export const generateId = (): string => {
  return `${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
}

/**
 * 防抖函数
 */
export const debounce = <T extends (...args: any[]) => any>(
  fn: T,
  delay: number
): ((...args: Parameters<T>) => void) => {
  let timer: ReturnType<typeof setTimeout>
  return (...args: Parameters<T>) => {
    clearTimeout(timer)
    timer = setTimeout(() => fn(...args), delay)
  }
}

/**
 * AI 错误消息分类
 */
export const getAIErrorMessage = (error: any): string => {
  const msg = error?.message || ''
  const status = error?.response?.status

  if (status === 401 || status === 403 || msg.includes('api key')) {
    return 'AI API 密钥无效，请联系管理员'
  }
  if (status === 429) {
    return 'AI API 请求过于频繁，请稍后重试'
  }
  if (msg.includes('timeout') || msg.includes('ETIMEDOUT')) {
    return '识别超时（超过 120 秒），请检查网络后重试'
  }
  if (msg.includes('Network Error')) {
    return '网络连接失败，请检查网络设置'
  }
  return msg || '识别失败，请稍后重试'
}

/**
 * 文件类型校验
 */
export const validateImageFile = (file: File): { valid: boolean; message?: string } => {
  const allowedTypes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']
  const maxSize = 10 * 1024 * 1024 // 10MB

  if (!allowedTypes.includes(file.type)) {
    return { valid: false, message: '仅支持 JPG、PNG、GIF、WebP 格式' }
  }
  if (file.size > maxSize) {
    return { valid: false, message: '文件大小不能超过 10MB' }
  }
  return { valid: true }
}
```
